API Layer

Endpoint บาง ๆ ส่งต่อให้ Use Case

public static class OrderEndpoints
{
    public static void MapOrderEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/orders")
                       .WithTags("Orders")
                       .RequireAuthorization();

        group.MapGet("/{id:guid}", async (Guid id, IOrderService svc, CancellationToken ct) =>
            await svc.GetByIdAsync(id, ct) is { } order
                ? Results.Ok(order)
                : Results.NotFound());

        group.MapPost("/", async (CreateOrderRequest req, IOrderService svc, CancellationToken ct) =>
        {
            var result = await svc.CreateAsync(req, ct);
            return result.IsSuccess
                ? Results.Created($"/api/orders/{result.Value.Id}", result.Value)
                : result.ToProblem();
        });
    }
}
Endpoint มีหน้าที่แค่ รับ request → เรียก service → แปลงผลเป็น HTTP response ห้ามมี business logic ในนี้ (Controllers ก็ใช้หลักเดียวกัน)

Transient

service เบา ๆ ไม่มี state เช่น mapper, calculator

Scoped

Async

async / await ที่ถูกต้อง

ห้ามทำทำแบบนี้แทนเหตุผล
.Result / .Wait()await ตลอดสายthread pool starvation / deadlock
async voidasync Taskexception หลุด จับไม่ได้ แอปล่ม
ไม่ส่ง CancellationTokenรับ ct แล้วส่งต่อถึง EF / HttpClientผู้ใช้ปิดหน้า query ก็หยุดตาม
new HttpClient() ทุกครั้งIHttpClientFactory / typed clientsocket exhaustion
ใช้ DbContext เดียวกันแบบขนานawait ทีละ query หรือแยก contextDbContext ไม่ thread-safe
Validation

Validate ด้วย FluentValidation

public sealed record CreateOrderRequest(
    Guid CustomerId,
    List<OrderItemRequest> Items,
    string? Note);

public sealed class CreateOrderValidator
    : AbstractValidator<CreateOrderRequest>
{
    public CreateOrderValidator()
    {
        RuleFor(x => x.CustomerId).NotEmpty();
        RuleFor(x => x.Items).NotEmpty()
            .WithMessage("ต้องมีสินค้าอย่างน้อย 1 รายการ");
        RuleForEach(x => x.Items).ChildRules(i =>
        {
            i.RuleFor(x => x.Quantity).InclusiveBetween(1, 999);
        });
        RuleFor(x => x.Note).MaximumLength(500);
    }
}

Checklist

Checklist ก่อน Merge API

  • ☐ Endpoint มี [Authorize] / policy ถูกต้อง
  • ☐ Input มี Validator, ตอบ 400 เป็น ProblemDetails
  • ☐ ส่ง CancellationToken ถึง EF Core
  • ☐ Query อ่านอย่างเดียวใช้ AsNoTracking + projection
  • ☐ list endpoint มี pagination
  • ☐ Migration ถูก review และ rollback ได้
  • ☐ OpenAPI (Swagger/Scalar) อัปเดต, ฝั่ง Next.js regenerate client
  • ☐ ไม่มี secret ใน appsettings / log
  • ☐ Unit + Integration test ผ่านใน CI
  • ☐ dotnet format และไม่มี warning
