Components

Component ที่ดีหน้าตาเป็นอย่างไร

type UserCardProps = {
  user: User;
  onSelect?: (id: string) => void;
};

export function UserCard({ user, onSelect }: UserCardProps) {
  return (
    <button className="card" onClick={() => onSelect?.(user.id)}>
      <Avatar src={user.avatarUrl} />
      <span>{user.fullName}</span>
    </button>
  );
}

ควรทำ

  • ทำหน้าที่เดียว (Single Responsibility)
  • Props มี type ชัดเจน ตั้งชื่อ event เป็น onXxx
  • ไฟล์ละ 1 component หลัก ยาวไม่เกิน ~200 บรรทัด
  • ใช้ named export เพื่อให้ refactor และค้นหาง่าย

ไม่ควรทำ

  • Component เดียวทั้ง fetch + จัดการ form + render ตาราง
  • ส่ง props ทะลุหลายชั้น (prop drilling) เกิน 3 ชั้น
  • ใช้ any กับ props

Hooks

Hooks ที่ใช้บ่อย และใช้เมื่อไร

useState

ค่าที่เปลี่ยนแล้วหน้าจอต้องเปลี่ยนตาม

state ซับซ้อน มีหลาย action เช่น wizard หลายขั้น

useMemo / useCallback

useActionState / useOptimistic

Pitfall

คุณอาจไม่ต้องใช้ useEffect

คำนวณค่าด้วย effect

const [items, setItems] = useState<Item[]>([]);
const [total, setTotal] = useState(0);

useEffect(() => {
  setTotal(items.reduce((s, i) => s + i.price, 0));
}, [items]); // render 2 รอบ + เสี่ยงไม่ sync

คำนวณระหว่าง render

const [items, setItems] = useState<Item[]>([]);

// derived state: คำนวณตรง ๆ ได้เลย
const total = items.reduce((s, i) => s + i.price, 0);

❌ fetch ข้อมูลใน effect

ใช้ TanStack Query หรือ Server Component แทน

❌ reset state เมื่อ props เปลี่ยน

ใช้ key แทน เช่น <Form key={userId} />

❌ ส่ง event ให้ parent ใน effect

เรียก callback ตรงใน event handler

Reuse Logic

แยก Logic ออกมาเป็น Custom Hook

เมื่อ logic เดียวกันถูกใช้มากกว่า 1 ที่ หรือ component เริ่มยาว ให้แยกเป็น hook ที่ขึ้นต้นด้วย use

// hooks/useDebounce.ts
export function useDebounce<T>(value: T, delay = 400) {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(t); // cleanup ทุกครั้ง
  }, [value, delay]);
  return debounced;
}

// ใช้งาน
const [keyword, setKeyword] = useState("");
const q = useDebounce(keyword);
const { data } = useProducts({ q });
Server State

ดึงข้อมูลด้วย TanStack Query

// features/products/api.ts
export const productKeys = {
  all: ["products"] as const,
  list: (f: ProductFilter) => [...productKeys.all, "list", f] as const,
};

export function useProducts(filter: ProductFilter) {
  return useQuery({
    queryKey: productKeys.list(filter),
    queryFn: () => api.get<Product[]>("/products", { params: filter }),
    staleTime: 60_000,
  });
}

export function useCreateProduct() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (dto: CreateProductDto) => api.post("/products", dto),
    onSuccess: () => qc.invalidateQueries({ queryKey: productKeys.all }),
  });
}

ได้อะไรฟรี

  • Cache + dedupe request
  • loading / error state
  • retry, refetch เมื่อกลับมาที่แท็บ
  • invalidate หลัง mutation

มาตรฐานทีม

  • รวม query key ไว้ที่ไฟล์เดียวต่อ feature
  • ห้ามเรียก fetch ตรงใน component
  • ใช้ API client กลางที่แนบ token / จัดการ error
Forms & Validation

Form = React Hook Form + Zod

const schema = z.object({
  email: z.string().email("อีเมลไม่ถูกต้อง"),
  password: z.string().min(8, "อย่างน้อย 8 ตัวอักษร"),
});
type LoginForm = z.infer<typeof schema>; // type มาจาก schema เดียว

export function LoginForm() {
  const { register, handleSubmit, formState: { errors, isSubmitting } } =
    useForm<LoginForm>({ resolver: zodResolver(schema) });

  return (
    <form onSubmit={handleSubmit(login)}>
      <input {...register("email")} />
      {errors.email && <p>{errors.email.message}</p>}
      <button disabled={isSubmitting}>เข้าสู่ระบบ</button>
    </form>
  );
}
Validate ที่ฝั่ง client เพื่อ UX แต่ ต้อง validate ซ้ำที่ .NET API เสมอ (FluentValidation) เพราะ client ถูก bypass ได้
Performance

ทำให้เร็ว โดยไม่ optimize ก่อนเวลา

  1. วัดก่อนแก้ ด้วย React DevTools Profiler และ Lighthouse
  2. ย้าย state ลงไปใกล้จุดที่ใช้ ลดจำนวน component ที่ re-render
  3. ใส่ key ที่ stable ใน list (ใช้ id ไม่ใช้ index)
  4. Lazy load หน้า/component ใหญ่ด้วย lazy() + Suspense
  5. Virtualize list ที่ยาวเกิน ~200 แถว (TanStack Virtual)
  6. เปิด React Compiler แทนการเขียน memo เองทุกที่

ทำไมแบบ Feature

แก้ feature เดียวอยู่ในโฟลเดอร์เดียว ลบ feature ทิ้งได้ง่าย และคนใหม่หาไฟล์เจอเร็ว

กฎการ import

  • features/a ห้าม import ไส้ใน features/b
  • ของที่ใช้ร่วมให้ย้ายไป components/ui หรือ lib
  • ใช้ alias @/ แทน ../../..

Checklist

Checklist ก่อนเปิด Pull Request

  • ☐ ไม่มี any, ไม่มี console.log หลงเหลือ
  • ☐ ไม่มี warning ใน console (key, act, hydration)
  • ☐ จัดการครบ 4 สถานะ: loading / error / empty / success
  • ☐ ปุ่ม submit กันกดซ้ำ (disabled ระหว่าง pending)
  • ☐ รองรับมือถือ (ทดสอบที่ 375px)
  • ☐ ใช้ semantic HTML / label ครบ (Accessibility)
  • ☐ ESLint + Prettier + tsc --noEmit ผ่าน
  • ☐ มี test สำหรับ logic สำคัญ (Vitest / Testing Library)
