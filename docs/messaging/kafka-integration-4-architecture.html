Controller บาง → Service → Producer

// app/api/produce/route.ts — Controller
export async function POST(request: Request) {
  const parsed = CreateOrderInput.safeParse(await request.json());
  if (!parsed.success) {
    return NextResponse.json({ errors: parsed.error.flatten() }, { status: 400 });
  }
  const result = await orderService.createOrder(parsed.data);
  return NextResponse.json(result, { status: 202 }); // รับเรื่องแล้ว กำลังประมวลผล
}

// modules/orders/order.service.ts — Business logic
export function createOrderService(deps: {
  repo: OrderRepository;
  events: OrderEventPublisher;          // interface ไม่ใช่ KafkaJS
}) {
  return {
    async createOrder(input: CreateOrderInput) {
      const order = buildOrder(input);  // คำนวณราคา, ตรวจกฎ
      await deps.repo.save({ ...order, status: "PRODUCED" });
      await deps.events.orderCreated(order);
      return { orderId: order.orderId, status: "PRODUCED" };
    },
  };
}
ตอบ 202 Accepted แทน 200 เพราะ order ยังไม่เสร็จจริง ผลลัพธ์สุดท้ายดูได้จาก tracking / สถานะ order (ตอนที่ 3)

Kafka Adapter และ Worker ที่ใช้ Service ชุดเดียวกัน

// modules/orders/order.producer.ts — Kafka adapter
export const kafkaOrderEvents: OrderEventPublisher = {
  async orderCreated(order) {
    await producer.send({
      topic: "order-events",
      messages: [{
        key: order.orderId,
        value: JSON.stringify(order),
        headers: { "x-correlation-id": order.orderId, "x-event-type": "OrderCreated" },
      }],
    });
  },
};

// ประกอบตอนเริ่มแอป (composition root)
export const orderService = createOrderService({
  repo: dbOrderRepository,
  events: kafkaOrderEvents,
});

// workers/consumer-inventory.ts — Worker
eachMessage: async ({ topic, partition, message }) => {
  const track = createTracker("Inventory", { topic, partition, message });
  await track("RECEIVED");
  const order = safeParseOrder(message.value);
  if (!order.success) return sendToDlt(...);

  // withRetry = loop retry + backoff ของตอนที่ 2 ห่อเป็น helper ใน lib/retry.ts
  await withRetry(() => inventoryService.reserve(order.data), {
    onRetry: (attempt, err) => track("RETRYING", { attempt, errorReason: String(err) }),
  });
  await track("PROCESS_SUCCESS");
}

// test: ไม่ต้องมี Kafka

Roadmap ต่อยอด Demo สู่ Production

  1. ใส่ key: orderId ตอน publish ใน /api/produce ตอนที่ 1
  2. เพิ่ม x-correlation-id ใน headers และส่งต่อทุก worker ตอนที่ 3
  3. tracking log มี topic / partition / offset / key ใน appendLog ตอนที่ 3
  4. retry log เป็น field attempt, maxRetries + backoff/jitter ตอนที่ 2
  5. API ค้น tracking ตาม orderId (timeline) ตอนที่ 3
  6. offset / lag ใน /api/consumer-status ตอนที่ 3
  7. สถานะ order เป็น lifecycle จริง PRODUCED → INVENTORY_OK → PAYMENT_OK → NOTIFIED / FAILED / DLT
  8. ย้าย tracking log ไป database หรือ log platform + แยก service ตามตอนนี้
สรุปทั้งซีรีส์

Kafka Integration ใน 4 ตอน

1. เตรียมข้อมูล & Contract

ขอข้อมูลครบ 4 กลุ่ม, schema + validator, key, headers, SASL_SSL

2. Retry & DLT

retry เฉพาะ transient, backoff + jitter, จบที่ DLT, idempotent

3. Status & Tracking

Kafka ไม่มี status → tracking log + correlationId + lag

4. โครงสร้างโค้ด

Controller / Service / Adapter แยกกัน test ได้ reuse ได้
