Docker Compose
รันทั้งระบบด้วยคำสั่งเดียว

เขียนไฟล์เดียวเพื่อรัน Next.js + .NET API + PostgreSQL + Redis พร้อมกัน ให้ทุก service คุยกันได้ และรอกันอย่างถูกลำดับ

compose.yamlservicesdepends_onhealthcheckprofiles
คนใหม่ clone repo → docker compose up → เริ่มเขียนโค้ดได้ภายใน 10 นาที
ปัญหาที่แก้

จาก docker run ยาว ๆ หลายคำสั่ง → ไฟล์เดียว

ไม่มี Compose

docker network create shop
docker run -d --name db --network shop -e POSTGRES_PASSWORD=dev -v pgdata:/var/lib/postgresql/data postgres:17-alpine
docker run -d --name redis --network shop redis:7-alpine
docker run -d --name api --network shop -p 8080:8080 -e ConnectionStrings__Default=... shop-api:dev
# ... จำลำดับเอง พิมพ์ผิดง่าย

มี Compose

docker compose up -d      # เปิดทั้งระบบ
docker compose down       # ปิดทั้งระบบ

ทุกอย่างอยู่ใน compose.yaml ที่ commit ลง Git ทุกคนได้ระบบเหมือนกัน

โครงสร้างไฟล์

โครงสร้างของ compose.yaml

# compose.yaml  (ไม่ต้องใส่ version: แล้ว)
name: shop

services:            # container แต่ละตัว
  db:
    image: postgres:17-alpine
  api:
    build: ./src/Shop.Api      # build จาก Dockerfile
    ports: ["8080:8080"]
    depends_on: [db]

volumes:             # ที่เก็บข้อมูลถาวร
  pgdata:

networks:            # (ไม่จำเป็น) Compose สร้าง network ให้อัตโนมัติ

service = 1 กล่องงาน

ชื่อ service (เช่น db) ใช้เป็น hostname คุยกันได้ทันที

image หรือ build

ใช้ image สำเร็จรูป หรือ build จาก Dockerfile ของเรา

ชื่อไฟล์

ใช้ compose.yaml (ชื่อใหม่) หรือ docker-compose.yml (ชื่อเดิม) ได้ทั้งคู่

ตัวอย่างจริง

Next.js + .NET API + PostgreSQL + Redis

name: shop
services:
  db:
    image: postgres:17-alpine
    environment:
      POSTGRES_DB: shop
      POSTGRES_USER: shop
      POSTGRES_PASSWORD: ${DB_PASSWORD}          # อ่านจากไฟล์ .env (ตอนที่ 4)
    volumes: [pgdata:/var/lib/postgresql/data]
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U shop -d shop"]
      interval: 5s
      retries: 10

  redis:
    image: redis:7-alpine

  api:
    build: { context: ., dockerfile: src/Shop.Api/Dockerfile }
    env_file: [./env/api.env]
    environment:
      ConnectionStrings__Default: Host=db;Database=shop;Username=shop;Password=${DB_PASSWORD}
      Redis__Connection: redis:6379
    ports: ["8080:8080"]
    depends_on:
      db: { condition: service_healthy }        # รอ DB พร้อมจริงก่อน
      redis: { condition: service_started }

  web:
    build: { context: ./web, args: { NEXT_PUBLIC_API_URL: http://localhost:8080 } }
    environment:
      API_URL: http://api:8080                  # ฝั่ง server ของ Next.js เรียกผ่านชื่อ service
    ports: ["3000:3000"]
    depends_on: [api]

volumes:
  pgdata:
ลำดับการเริ่ม

depends_on อย่างเดียว ไม่พอ

depends_on แบบธรรมดา

รอแค่ให้ container ของ DB “เริ่ม” แต่ PostgreSQL ยังเปิดไม่เสร็จ

→ API เชื่อมต่อ DB ไม่ได้ แล้วล่มตอนเปิด

depends_on + healthcheck

รอจน DB ตอบ pg_isready ผ่านจริง

→ API เปิดได้ทุกครั้ง

conditionรอจนกว่า
service_startedcontainer เริ่มทำงาน (ค่าเริ่มต้น)
service_healthyhealthcheck ผ่าน — ใช้กับ DB, Redis, message broker
service_completed_successfullycontainer จบด้วย exit 0 — ใช้กับงาน migration ที่ต้องเสร็จก่อน

แม้จะรอแล้ว แอปควรมี retry ตอนเชื่อมต่อ DB ด้วย เพราะใน production DB อาจหลุดได้ทุกเมื่อ

CLI

คำสั่ง Compose ที่ใช้ทุกวัน

docker compose up -d              # เปิดทั้งหมด (เบื้องหลัง)
docker compose up -d --build      # build ใหม่ก่อนเปิด (แก้ Dockerfile/โค้ด)
docker compose up -d api          # เปิดเฉพาะ api (+ ตัวที่มันพึ่ง)
docker compose ps                 # สถานะทุก service
docker compose logs -f api        # ดู log ของ api
docker compose exec db psql -U shop   # รันคำสั่งใน service
docker compose restart api
docker compose stop               # หยุด (ไม่ลบ)
docker compose down               # หยุด + ลบ container และ network
docker compose down -v            # ลบ volume ด้วย (ข้อมูล DB หาย!)
docker compose config             # ดูไฟล์ที่รวมและแทนค่าตัวแปรแล้ว

docker compose config

คำสั่งที่ช่วย debug ได้มากที่สุด: เห็นค่าจริงหลังแทน ${VAR} และรวมหลายไฟล์แล้ว

down -v

ลบ volume = ลบข้อมูลฐานข้อมูล ใช้เมื่ออยากเริ่มใหม่จริง ๆ เท่านั้น

docker compose ไม่ใช่ docker-compose

v2 เป็น plugin ของ docker แล้ว ตัวมีขีด (v1) เลิกพัฒนาแล้ว

แยก dev / prod

แยกค่าตาม environment ด้วยหลายไฟล์

# compose.yaml            → ส่วนที่เหมือนกันทุก env
# compose.override.yaml   → dev: Compose อ่านอัตโนมัติ!
services:
  api:
    environment:
      ASPNETCORE_ENVIRONMENT: Development
    ports: ["8080:8080"]
  db:
    ports: ["5432:5432"]   # เปิดให้ DBeaver ต่อได้ (dev เท่านั้น)

# dev: compose.yaml + compose.override.yaml (อัตโนมัติ)
docker compose up -d

# prod: ระบุไฟล์เอง (ไม่อ่าน override)
docker compose -f compose.yaml -f compose.prod.yaml up -d
ไฟล์หลังจะ “ทับ” ค่าของไฟล์ก่อนหน้า ใช้แยกสิ่งที่ต่างกัน เช่น port ที่เปิด, environment, จำนวน replica
เปิดเฉพาะที่ต้องใช้

Profiles: service เสริมที่ไม่ต้องรันทุกครั้ง

services:
  db: { image: postgres:17-alpine }

  pgadmin:
    image: dpage/pgadmin4
    profiles: [tools]          # ไม่รันโดยค่าเริ่มต้น
    ports: ["5050:80"]

  mailpit:
    image: axllent/mailpit     # จับอีเมลทดสอบ ดูได้ที่ :8025
    profiles: [tools]
    ports: ["8025:8025", "1025:1025"]

docker compose up -d                     # db เท่านั้น
docker compose --profile tools up -d     # db + pgadmin + mailpit

ตัวอย่างที่ใช้บ่อย: เครื่องมือจัดการ DB, Mailpit สำหรับทดสอบอีเมล, Kafka UI / RabbitMQ management, job ที่ seed ข้อมูล

Pitfalls

ข้อผิดพลาดที่พบบ่อย

ใช้ localhost คุยข้าม service

ใน container localhost คือตัวมันเอง ต้องใช้ชื่อ service เช่น Host=db

แก้โค้ดแล้วไม่เห็นผล

image เดิมยังอยู่ ต้อง up -d --build (หรือใช้ watch ในตอนที่ 6)

port ซ้ำกับโปรแกรมในเครื่อง

"port is already allocated" → เปลี่ยนฝั่งซ้าย เช่น "15432:5432"

เปิด port DB ใน production

ให้ service คุยกันใน network ภายใน ไม่ต้อง ports: สำหรับ DB
